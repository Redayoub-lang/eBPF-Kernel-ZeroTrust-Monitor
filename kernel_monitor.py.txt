class PayloadEntropyAnalyzer:

  def __init__(self, entropy_threshold: float = 7.2):
    """
    :param entropy_threshold: Shannon entropy limit (bits/byte) triggering obfuscation alert.
    """
    self.threshold = entropy_threshold

  def compute_shannon_entropy(self, payload: bytes) -> float:
    """Computes exact Shannon Entropy H(X) = - sum(P(x) * log2(P(x))) over raw byte stream."""
    if not payload:
      return 0.0

    length = len(payload)
    counts = np.bincount(np.frombuffer(payload, dtype=np.uint8), minlength=256)
    probabilities = counts[counts > 0] / length

    entropy = -np.sum(probabilities * np.log2(probabilities))
    return float(entropy)

  def inspect_payload(
      self, pid: int, process_name: str, payload: bytes
  ) -> dict:
    """Inspects execution event telemetry for high-entropy shellcode injection."""
    entropy = self.compute_shannon_entropy(payload)
    is_anomaly = entropy > self.threshold

    return {
        "timestamp": time.time(),
        "pid": pid,
        "process": process_name,
        "payload_len": len(payload),
        "entropy": round(entropy, 4),
        "alert": is_anomaly,
    }
